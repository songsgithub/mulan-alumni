import type { CollectionConfig } from 'payload'

import { opsOnly, publicContentRead } from '../access'
import { generateSlug } from '../lib/slug'

/**
 * 可复用经验（借鉴 Ashoka 的经验沉淀机制）：
 * 把一个互助团跑通过的活动方法沉淀成公开可借鉴的工具包。
 */
export const Toolkits: CollectionConfig = {
  slug: 'toolkits',
  labels: {
    singular: '互助团经验',
    plural: '互助团经验',
  },
  admin: {
    group: '公开内容',
    useAsTitle: 'title',
    defaultColumns: ['title', 'cohort', 'reusable', 'publishedAt'],
  },
  access: {
    // 公开可读（仅公开可复用的条目）
    read: publicContentRead({ reusable: { equals: true } }),
    // 公开站内容：只由项目组维护（团长不能改公开页面上的内容）
    create: opsOnly,
    update: opsOnly,
    delete: opsOnly,
  },
  fields: [
    {
      name: 'title',
      type: 'text',
      label: '标题',
      required: true,
    },
    {
      name: 'slug',
      type: 'text',
      label: '地址标识',
      required: true,
      unique: true,
      index: true,
      admin: {
        description:
          '地址栏里的那一段（/资源路径/<标识>）。不透明随机串，不带关键词、拼音或城市。',
      },
    },
    {
      name: 'cohort',
      type: 'relationship',
      relationTo: 'cohorts',
      label: '来源互助团',
    },
    {
      name: 'summary',
      type: 'textarea',
      label: '一句话说明',
      required: true,
    },
    {
      name: 'body',
      type: 'richText',
      label: '正文',
    },
    {
      name: 'files',
      type: 'upload',
      relationTo: 'media',
      label: '附件（模板/清单）',
      hasMany: true,
    },
    {
      name: 'reusable',
      type: 'checkbox',
      label: '公开可复用',
      defaultValue: true,
    },
    {
      name: 'publishedAt',
      type: 'date',
      label: '发布时间',
      defaultValue: () => new Date().toISOString(),
    },
    {
      name: 'author',
      type: 'relationship',
      relationTo: 'users',
      label: '整理人',
      admin: {
        readOnly: true,
      },
    },
  ],
  hooks: {
    beforeChange: [
      ({ data, req }) => {
        if (!data?.author && req.user) {
          data.author = req.user.id
        }
        return data
      },
    ],
    // 地址标识统一是不透明随机串
    beforeValidate: [generateSlug('toolkits')],
  },
}
